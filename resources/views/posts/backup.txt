// DYNAMIC, CATEGORY-AWARE SEO
// Each category gets its own title suffix / fallback description /
// keyword set / schema.org type, so posts across very different verticals
// (AI, Finance, Health, Product Reviews, Local guides, etc.) each get

// instead of one generic "Laravel/PHP" fallback for every post.
// =====================================================================

$postTitle   = $cached['title'] ?? $post->title;
$postSlug    = $cached['slug'] ?? $post->slug;
$categoryName = $cached['category']['name'] ?? optional($post->category ?? null)->name ?? null;
$canonicalUrl = route('posts.show', $postSlug);

$categorySeoMap = [
    'Artificial Intelligence' => [
        'suffix'   => 'AI Insights & Guides',

        'keywords' => 'artificial intelligence, AI news, machine learning, AI tools, AI guide',
    ],
    'Finance & Online Earning' => [
        'suffix'   => 'Finance & Online Earning',
        'desc'     => 'Practical money management tips and proven, legitimate ways to earn income online.',
        'keywords' => 'personal finance, online earning, make money online, side income, financial tips',
    ],
    'Health & Fitness' => [
        'suffix'   => 'Health & Fitness Guide',
        'desc'     => 'Evidence-informed health and fitness guidance to help you build sustainable habits.',
        'keywords' => 'health tips, fitness guide, wellness, workout, healthy lifestyle',
    ],
    'Education & Career' => [
        'suffix'   => 'Education & Career Advice',
        'desc'     => 'Actionable education and career advice to help you plan your next academic or professional step.',
        'keywords' => 'career advice, education guide, job skills, career growth, study tips',
    ],
    'Tech Reviews' => [
        'suffix'   => 'Tech Review',
        'desc'     => 'Hands-on, honest technology reviews to help you make smarter buying decisions.',
        'keywords' => 'tech review, gadget review, product comparison, tech news',
    ],
    'Digital Marketing & SEO' => [
        'suffix'   => 'Digital Marketing & SEO Tips',
        'desc'     => 'Actionable digital marketing and SEO strategies to help you grow traffic, rankings, and conversions.',
        'keywords' => 'digital marketing, SEO tips, content marketing, search engine optimization',
    ],
    'Product Review (Affiliate)' => [
        'suffix'   => 'Honest Product Review',
        'desc'     => 'An honest, detailed product review covering features, pros and cons, and value for money.',
        'keywords' => 'product review, buying guide, best products, honest review',
    ],
    'Local Informational' => [
        'suffix'   => 'Local Guide',
        'desc'     => 'Local, region-specific information, guides, and recommendations.',
        'keywords' => 'local guide, travel tips, city guide, local information',
    ],
];

$catMeta = $categorySeoMap[$categoryName] ?? [
    'suffix'   => 'Article',
    'desc'     => null,
    'keywords' => null,
];

// Title: keep it natural for short titles, but always end with a category
// suffix + brand so every post has a distinct, descriptive <title>.
$seoTitle = $postTitle . ' | ' . $catMeta['suffix'] . ' | NSential';

// Description: prefer real content, fall back to category-specific copy
// instead of one generic Laravel/PHP blurb for every vertical.
$contentText = trim(strip_tags($cached['content'] ?? $post->content ?? ''));
if (strlen($contentText) >= 50) {
    $ogDescription = Str::limit($contentText, 160);
} elseif ($catMeta['desc']) {
    $ogDescription = 'Read "' . $postTitle . '" — ' . $catMeta['desc'];
} else {
    $ogDescription = 'Read "' . $postTitle . '" on NSential.';
}
$ogDescription = Str::limit($ogDescription, 160, '');

$ogImage = isset($cached['featured_image']) && $cached['featured_image']
    ? $cached['featured_image']
    : asset('images/default-og-image.jpg');

// Product reviews get og:type=product (and Product schema below);
// everything else stays a standard article.
$isProductReview = $categoryName === 'Product Review (Affiliate)';
$ogType = $isProductReview ? 'product' : 'article';

$postStatus = $cached['status'] ?? $post->status;
$isPublished = $postStatus === 'published';

$reviewsCount = (int) ($cached['reviews_count'] ?? 0);
$averageRating = (float) ($cached['average_rating'] ?? 0);

$publishedIso = (isset($cached['published_at_raw']) && $cached['published_at_raw'])
    ? \Carbon\Carbon::parse($cached['published_at_raw'])->toIso8601String()
    : null;
$modifiedIso = isset($post->updated_at) ? \Carbon\Carbon::parse($post->updated_at)->toIso8601String() : $publishedIso;

// ---- JSON-LD structured data ----
$jsonLd = [
    '@context' => 'https://schema.org',
    '@graph' => [],
];

// Breadcrumbs
$breadcrumbItems = [
    ['@type' => 'ListItem', 'position' => 1, 'name' => 'Home', 'item' => url('/')],
];
if ($categoryName) {
    $breadcrumbItems[] = [
        '@type' => 'ListItem',
        'position' => 2,
        'name' => $categoryName,
        'item' => url('/categories/' . ($cached['category']['slug'] ?? Str::slug($categoryName))),
    ];
}
$breadcrumbItems[] = [
    '@type' => 'ListItem',
    'position' => count($breadcrumbItems) + 1,
    'name' => $postTitle,
    'item' => $canonicalUrl,
];
$jsonLd['@graph'][] = [
    '@type' => 'BreadcrumbList',
    'itemListElement' => $breadcrumbItems,
];

// Core Article (or Product, for affiliate reviews) node
$articleNode = [
    '@type' => $isProductReview ? 'Product' : 'Article',
    'name' => $postTitle,
    'headline' => Str::limit($postTitle, 110, ''),
    'description' => $ogDescription,
    'image' => $ogImage,
    'url' => $canonicalUrl,
    'mainEntityOfPage' => ['@type' => 'WebPage', '@id' => $canonicalUrl],
];

if (!$isProductReview) {
    $articleNode['@type'] = 'Article';
    $articleNode['articleSection'] = $categoryName;
    if ($publishedIso) $articleNode['datePublished'] = $publishedIso;
    if ($modifiedIso) $articleNode['dateModified'] = $modifiedIso;
    if (isset($cached['author']) && $cached['author']) {
        $articleNode['author'] = [
            '@type' => 'Person',
            'name' => $cached['author']['name'],
        ];
    }
    $articleNode['publisher'] = [
        '@type' => 'Organization',
        'name' => 'NSential',
        'logo' => ['@type' => 'ImageObject', 'url' => asset('images/logo.png')],
    ];
}

// Ratings/reviews apply to both content types when present, so posts with
// user reviews (e.g. product reviews, tool comparisons) get eligible for
// star-rating rich snippets in search results.
if ($reviewsCount > 0 && $averageRating > 0) {
    $articleNode['aggregateRating'] = [
        '@type' => 'AggregateRating',
        'ratingValue' => number_format($averageRating, 1),
        'reviewCount' => $reviewsCount,
        'bestRating' => 5,
        'worstRating' => 1,
    ];
}

if ($isProductReview) {
    // Product schema requires an "offers" or "review" node to be fully
    // valid for rich results; brand/offer details should be filled in
    // from post/product meta once that data exists on the model.
    if (isset($cached['author']) && $cached['author']) {
        $articleNode['review'] = [
            '@type' => 'Review',
            'author' => ['@type' => 'Person', 'name' => $cached['author']['name']],
            'reviewRating' => [
                '@type' => 'Rating',
                'ratingValue' => $averageRating > 0 ? number_format($averageRating, 1) : 5,
                'bestRating' => 5,
                'worstRating' => 1,
            ],
        ];
    }
}

$jsonLd['@graph'][] = $articleNode;

    function showToast(message, type = 'info') {
        const container = document.getElementById('toastContainer');
        const toast = document.createElement('div');
        toast.className = `toast ${type}`;
        toast.textContent = message;
        container.appendChild(toast);
        setTimeout(() => toast.classList.add('show'), 10);
        setTimeout(() => {
            toast.classList.remove('show');
            setTimeout(() => toast.remove(), 300);
        }, 3000);
    }

    // ===== CONFIRM FUNCTION =====
    function customConfirm(message) {
        return new Promise((resolve) => {
            const modal = document.getElementById('confirmModal');
            document.getElementById('confirmMessage').textContent = message;
            modal.classList.add('open');
            
            function cleanup() {
                modal.classList.remove('open');
                document.removeEventListener('keydown', onKeydown);
            }
            
            function onKeydown(e) {
                if (e.key === 'Escape') { cleanup(); resolve(false); }
                if (e.key === 'Enter') { cleanup(); resolve(true); }
            }
            
            document.getElementById('confirmOk').onclick = () => { cleanup(); resolve(true); };
            document.getElementById('confirmCancel').onclick = () => { cleanup(); resolve(false); };
            document.addEventListener('keydown', onKeydown);
            setTimeout(() => document.getElementById('confirmOk').focus(), 100);
        });
    }

    document.addEventListener('DOMContentLoaded', function() {
        const commentForm = document.getElementById('commentForm');

        const postId = {{ $post->id }};

        // ========== HANDLE REPLY SUBMISSION ==========
        function handleReplySubmission(replyForm) {
            const replyText = replyForm.querySelector('.reply-textarea').value;
            const commentId = replyForm.dataset.commentId;

            if (!replyText || replyText.trim().length < 2) {
                showToast('Please write a reply (minimum 2 characters).', 'warning');
                return;
            }

            const submitBtn = replyForm.querySelector('.submit-reply-btn');
            const originalText = submitBtn.textContent;
            submitBtn.disabled = true;
            submitBtn.textContent = 'Replying...';

            fetch(`/posts/${postId}/comments`, {
                method: 'POST',
                headers: {
                    'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content,
                    'Accept': 'application/json',
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    content: replyText,
                    parent_comment_id: commentId
                })
            })
            .then(response => {
                if (!response.ok) {
                    return response.json().then(err => { throw new Error(err.errors?.content?.[0] || 'Failed to post reply'); });
                }
                return response.json();
            })
            .then(data => {
                const commentItem = replyForm.closest('.comment-item');
                const repliesContainer = commentItem.querySelector('.replies-container');

                if (repliesContainer.children.length === 0) {
                    repliesContainer.className = 'replies-container mt-2 space-y-2 pl-3 border-l border-hairline';
                }

                const replyHtml = createCommentHtml(data.comment, true);
                repliesContainer.insertAdjacentHTML('beforeend', replyHtml);

                replyForm.querySelector('.reply-textarea').value = '';
                replyForm.classList.add('hidden');

                const count = parseInt(commentCount.textContent) + 1;
                commentCount.textContent = count;
                if (commentCountDrawer) commentCountDrawer.textContent = count;

                const successMsg = document.createElement('div');
                successMsg.className = 'text-cyan text-[11px] mt-1';
                successMsg.textContent = '✓ Reply posted!';
                repliesContainer.appendChild(successMsg);
                setTimeout(() => successMsg.remove(), 2000);
            })
            .catch(error => {
                showToast(error.message || 'Failed to post reply. Please try again.', 'error');
            })
            .finally(() => {
                submitBtn.disabled = false;
                submitBtn.textContent = originalText;
            });
        }

                        },
                        body: JSON.stringify({
                            content: content
                        })
                    });

                    if (!response.ok) {
                        const errorData = await response.json();
                        throw new Error(errorData.errors?.content?.[0] || 'Failed to post comment');
                    }

                    const data = await response.json();

                    const noComments = document.getElementById('noComments');
                    if (noComments) {
                        noComments.remove();
                    }

                    const newCommentHtml = createCommentHtml(data.comment);
                    commentsList.insertAdjacentHTML('afterbegin', newCommentHtml);

                    const count = parseInt(commentCount.textContent) + 1;
                    commentCount.textContent = count;
                    if (commentCountDrawer) commentCountDrawer.textContent = count;

                    document.getElementById('commentContent').value = '';

                    const successMsg = document.createElement('div');
                    successMsg.className = 'text-cyan text-xs mt-2';
                    successMsg.textContent = '✓ Comment posted successfully!';
                    this.appendChild(successMsg);
                    setTimeout(() => successMsg.remove(), 3000);

                } catch (error) {
                    showToast(error.message || 'Failed to post comment. Please try again.', 'error');
                } finally {
                    submitBtn.disabled = false;
                    submitBtn.textContent = originalText;
                }
            });
        }

        function updateReviewDisplay(data) {

                avgDisplay.textContent = avgRating.toFixed(1);

                totalDisplay.textContent = (data.total_reviews || 0) + ' reviews';
            }

            const openReviewsBtnSpans = openReviewsBtn ? openReviewsBtn.querySelectorAll('span') : [];
            if (openReviewsBtnSpans.length >= 2) {
                const avgRating = parseFloat(data.average_rating) || 0;
                openReviewsBtnSpans[0].textContent = avgRating.toFixed(1);
                openReviewsBtnSpans[1].textContent = `(${data.total_reviews || 0} reviews)`;
            }

            const starDisplay = document.getElementById('starDisplay');
            if (starDisplay) {
                const avg = Math.round(parseFloat(data.average_rating) || 0);
                starDisplay.innerHTML = '';
                for (let i = 1; i <= 5; i++) {
                    const star = document.createElement('i');
                    star.className = `fas fa-star ${i <= avg ? 'text-amber-400' : 'text-mist/30'}`;
                    starDisplay.appendChild(star);
                }
            }

            const distribution = data.rating_distribution || {};
            const total = parseInt(data.total_reviews) || 0;
            const bars = document.querySelectorAll('.dist-bar');
            const counts = document.querySelectorAll('.dist-count');

            let index = 0;
            for (let i = 5; i >= 1; i--) {
                const count = parseInt(distribution[i]) || 0;
                const percentage = total > 0 ? (count / total * 100) : 0;
                if (bars[index]) {
                    bars[index].style.width = percentage + '%';
                }
                if (counts[index]) {
                    counts[index].textContent = count;
                }
                index++;
            }
        }

    (function() {
        const postId = {{ $post->id }};
        const csrfToken = document.querySelector('meta[name="csrf-token"]')?.content;
        if (!csrfToken) return;

        let lastActiveCheck = Date.now();
        let accumulatedSeconds = 0;
        let isPageVisible = !document.hidden;
        const FLUSH_INTERVAL_MS = 60000;

        function tick() {
            if (isPageVisible) {
                const now = Date.now();
                accumulatedSeconds += Math.floor((now - lastActiveCheck) / 1000);
                lastActiveCheck = now;
            }
        }

        function flush(useBeacon = false) {
            tick();
            if (accumulatedSeconds < 1) return;

            const duration = accumulatedSeconds;
            accumulatedSeconds = 0;

            if (useBeacon && navigator.sendBeacon) {
                const formData = new FormData();
                formData.append('duration', duration);
                navigator.sendBeacon(`/posts/${postId}/track-read`, formData);
                return;
            }

            fetch(`/posts/${postId}/track-read`, {
                method: 'POST',
                headers: {
                    'X-CSRF-TOKEN': csrfToken,
                    'Accept': 'application/json',
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({ duration }),
                keepalive: true,
            }).then(r => r.json()).then(data => {
                if (data.success) updateDisplay(data.new_value);
            }).catch(() => {});
        }

        function updateDisplay(seconds) {
            const display = document.getElementById('readingTimeDisplay');
            if (display && seconds) {
                display.textContent = `${Math.ceil(seconds / 60)} min read`;
            }
        }

        document.addEventListener('visibilitychange', function() {
            tick();
            isPageVisible = !document.hidden;
            lastActiveCheck = Date.now();
        });

        setInterval(() => flush(false), FLUSH_INTERVAL_MS);
        window.addEventListener('pagehide', () => flush(true));
    })();
